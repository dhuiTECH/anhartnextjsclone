// Simplified Google Apps Script for Anhart Form Submissions
// Handles form submissions and saves them to Google Sheets

const ALLOWED_ORIGINS = [
  'https://www.anhart.ca',
  'https://www.staging.anhart.ca',
  'https://20e626fa-0378-4012-9cbe-6de81c265661.lovableproject.com', // Add this for testing
  'https://anhartca.lovable.app'
];

function doGet(e) {
  return doPost(e); // Google Apps Script web apps require doGet for deployment
}

function doPost(e) {
  try {
    // Get actual origin from request headers
    const origin = e.parameter.origin || '';
    
    let data;
    if (e.postData && e.postData.type === "application/json") {
      data = JSON.parse(e.postData.contents);
    } else {
      data = e.parameter;
    }

    // Validate submission
    const securityCheck = validateSubmission(data);
    if (!securityCheck.valid) {
      return createJsonResponse(
        { success: false, error: "Invalid submission: " + securityCheck.reason },
        origin
      );
    }

    // Check rate limit
    if (!checkRateLimit(data.email)) {
      return createJsonResponse(
        { success: false, error: "Rate limit exceeded. Please try again later." },
        origin
      );
    }

    // Save to sheet
    saveToSheet(data);

    // Send emails
    sendConfirmationEmail(data);
    sendAdminNotification(data);

    return createJsonResponse(
      { success: true, message: "Form submitted successfully" },
      origin
    );

  } catch (error) {
    return createJsonResponse(
      { success: false, error: error.message },
      e.parameter.origin || ''
    );
  }
}

function doOptions(e) {
  const origin = e.parameter.origin || (e.headers && e.headers.origin);
  const output = ContentService.createTextOutput("")
    .setMimeType(ContentService.MimeType.TEXT);

  if (origin && ALLOWED_ORIGINS.includes(origin)) {
    output.addHeader("Access-Control-Allow-Origin", origin);
    output.addHeader("Access-Control-Allow-Methods", "POST, OPTIONS");
    output.addHeader("Access-Control-Allow-Headers", "Content-Type");
  }

  return output;
}

function createJsonResponse(payload, origin) {
  const output = ContentService.createTextOutput(JSON.stringify(payload))
    .setMimeType(ContentService.MimeType.JSON);
  
  // Add CORS headers if origin is allowed
  if (origin && ALLOWED_ORIGINS.includes(origin)) {
    output.addHeader('Access-Control-Allow-Origin', origin);
    output.addHeader('Access-Control-Allow-Methods', 'POST, OPTIONS');
    output.addHeader('Access-Control-Allow-Headers', 'Content-Type');
  }
  
  return output;
}

function saveToSheet(data) {
  const sheet = SpreadsheetApp.getActiveSheet();
  const timestamp = new Date();
  
  sheet.appendRow([
    timestamp,
    data.name || '',
    data.email || '',
    data.phone || '',
    data.organization || '',
    data.subject || '',
    data.message || '',
    data.form_type || '',
    data.investment_amount || '',
    data.user_id || '',
    data.userAgent || '',
    data.referrer || ''
  ]);
}

function validateSubmission(data) {
  if (!data.name || data.name.trim().length < 2) {
    return { valid: false, reason: 'Invalid name' };
  }
  
  if (!data.email || !isValidEmail(data.email)) {
    return { valid: false, reason: 'Invalid email' };
  }
  
  if (!data.message || data.message.trim().length < 10) {
    return { valid: false, reason: 'Message too short' };
  }

  const suspiciousPatterns = [
    /<script/i,
    /javascript:/i,
    /on\w+\s*=/i,
    /href\s*=\s*["']javascript:/i,
    /<iframe/i,
    /<object/i,
    /<embed/i
  ];

  const allText = `${data.name} ${data.email} ${data.message}`.toLowerCase();
  for (const pattern of suspiciousPatterns) {
    if (pattern.test(allText)) {
      return { valid: false, reason: 'Suspicious content detected' };
    }
  }

  if (data.message.length > 5000) {
    return { valid: false, reason: 'Message too long' };
  }
  
  return { valid: true };
}

function isValidEmail(email) {
  return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email);
}

function checkRateLimit(email) {
  const key = 'rate_' + email.replace(/[^a-zA-Z0-9]/g, '');
  const props = PropertiesService.getScriptProperties();
  const data = props.getProperty(key);

  if (data) {
    const { count, timestamp } = JSON.parse(data);
    const oneHourAgo = Date.now() - (60 * 60 * 1000);
    
    if (timestamp > oneHourAgo && count >= 3) {
      return false;
    }
  }

  const newCount = data ? JSON.parse(data).count + 1 : 1;
  props.setProperty(key, JSON.stringify({
    count: newCount,
    timestamp: Date.now()
  }));

  return true;
}

function getFormTypeDescription(formType) {
  const descriptions = {
    'home': 'contact',
    'contact': 'contact',
    'limited_partnership': 'limited partnership investment',
    'partner': 'partnership'
  };
  return descriptions[formType] || 'form';
}

function setupSheet() {
  const sheet = SpreadsheetApp.getActiveSheet();
  const headers = [
    'Timestamp', 'Name', 'Email', 'Phone', 'Organization', 'Subject', 'Message',
    'Form Type', 'Investment Amount', 'User ID', 'User Agent', 'Referrer'
  ];
  sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
  sheet.getRange(1, 1, 1, headers.length).setFontWeight('bold');
  sheet.getRange(1, 1, 1, headers.length).setBackground('#D32F2F');
  sheet.getRange(1, 1, 1, headers.length).setFontColor('white');
}
